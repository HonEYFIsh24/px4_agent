cmake_minimum_required(VERSION 3.10)
project(uav_body_imu_bridge)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs)
find_package(gazebo REQUIRED)
find_path(GAZEBO_PROTO_INCLUDE quaternion.pb.h PATHS ${GAZEBO_INCLUDE_DIRS} PATH_SUFFIXES gazebo/msgs)
if(DEFINED ENV{PX4_DIR})
  set(PX4_DEFAULT_DIR "$ENV{PX4_DIR}")
else()
  set(PX4_DEFAULT_DIR "$ENV{HOME}/PX4-Autopilot")
endif()
set(PX4_GAZEBO_BUILD "${PX4_DEFAULT_DIR}/build/px4_sitl_default/build_gazebo-classic" CACHE PATH "PX4 Gazebo message library directory")
catkin_package()
add_library(uav_body_imu_plugin SHARED src/body_imu_plugin.cpp)
target_compile_features(uav_body_imu_plugin PRIVATE cxx_std_17)
target_compile_options(uav_body_imu_plugin PRIVATE -Wall -Wextra)
target_include_directories(uav_body_imu_plugin PRIVATE ${catkin_INCLUDE_DIRS} ${GAZEBO_INCLUDE_DIRS} ${GAZEBO_PROTO_INCLUDE} "${PX4_GAZEBO_BUILD}")
target_link_libraries(uav_body_imu_plugin ${catkin_LIBRARIES} ${GAZEBO_LIBRARIES} "${PX4_GAZEBO_BUILD}/libsensor_msgs.so")
